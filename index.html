// ฟังก์ชันดึงข้อมูลราคาของจริงผ่าน API async function fetchPriceData(type, name) { if (type === 'gold') { // ดึงราคาทองจาก API สมาคมค้าทองคำ try { const response = await fetch('https://api.chnwt.dev/thai-gold-api/latest'); const data = await response.json(); const goldBuy = data.response.price.gold_bar.buy; const goldSell = data.response.price.gold_bar.sell; const apiTime = data.response.update_time; // สร้างวันที่เองในรูปแบบสั้น (เช่น 13 ส.ค. 69) const currentDate = new Date().toLocaleDateString('th-TH', { day: 'numeric', month: 'short', year: '2-digit' }); return { title: "ราคาทองคำแท่ง", line1: `ขายออก : ${goldSell}`, line2: `รับซื้อ : ${goldBuy}`, line3: `อัปเดต : ${currentDate} เวลา ${apiTime}`, color: "#FFB000", size1: "xxl", // แก้ไขจาก 2xl เป็น xxl size2: "xxl", // แก้ไขจาก 2xl เป็น xxl weight2: "bold", color2: "#333333" }; } catch (error) { console.error("Gold API Error:", error); return { title: "ราคาทองคำแท่ง", line1: "ไม่สามารถดึงข้อมูลได้", line2: "-", line3: "-", color: "#FF334B", size1: "xl", size2: "md", weight2: "regular", color2: "#666666" }; } } else { // ดึงราคาหุ้นไทยผ่าน Yahoo Finance API และ Proxy try { const symbol = name.toUpperCase() + '.BK'; const url = `https://query1.finance.yahoo.com/v8/finance/chart/${symbol}`; const proxyUrl = `https://api.allorigins.win/get?url=${encodeURIComponent(url)}`; const response = await fetch(proxyUrl); const proxyData = await response.json(); const data = JSON.parse(proxyData.contents); const result = data.chart.result[0]; const meta = result.meta; const currentPrice = meta.regularMarketPrice; const previousClose = meta.chartPreviousClose; // คำนวณส่วนต่าง และ % การเปลี่ยนแปลง const change = currentPrice - previousClose; const percentChange = ((change / previousClose) * 100).toFixed(2); const sign = change > 0 ? "+" : (change < 0 ? "-" : ""); const color = change > 0 ? "#1DB446" : (change < 0 ? "#FF334B" : "#999999"); return { title: `ราคาหุ้น ${name.toUpperCase()}`, line1: `${currentPrice.toFixed(2)} ฿`, line2: `เปลี่ยนแปลง : ${sign}${Math.abs(change).toFixed(2)}`, line3: `คิดเป็น : ${sign}${Math.abs(percentChange)}%`, color: color, size1: "xxl", // แก้ไขจาก 2xl เป็น xxl size2: "md", weight2: "regular", color2: "#666666" }; } catch (error) { console.error("Stock API Error:", error); return { title: `ไม่พบหุ้น ${name.toUpperCase()}`, line1: "ไม่พบข้อมูล", line2: "โปรดตรวจสอบชื่อหุ้นอีกครั้ง", line3: "-", color: "#FF334B", size1: "xl", size2: "md", weight2: "regular", color2: "#666666" }; } } }
